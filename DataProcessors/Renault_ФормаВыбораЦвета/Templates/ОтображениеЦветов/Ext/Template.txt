<script type="text/javascript">

	if(!document.getElementById)document.getElementById=function(elemID){return eval("document.all['" + elemID + "']");}

	function rgb2hex(r,g,b){
		return (
			(0x100 | r).toString(16).substr(1)+
			(0x100 | g).toString(16).substr(1)+
			(0x100 | b).toString(16).substr(1));
	}

	function setColor(element,r,g,b){
		element.style.backgroundColor='#'+rgb2hex(r,g,b);
		element.style.color=0.213*r+0.715*g+0.072*b<127.5 ? '#ffffff' : '#000000';
	}

	function setOldColor(r,g,b){setColor(document.getElementById("oldColor"),r,g,b);}

	function setNewColor(r,g,b){setColor(document.getElementById("newColor"),r,g,b);}

	function doLoad(){
		document.ondragstart = function(){return false};
		document.onselectstart = function(e){if(!e)e=window.event;e.returnValue=false;};
		document.ontextmenu = function(){return false};
		document.oncontextmenu = function(e){
			if(!e)e=window.event;
			if (e.preventDefault)
				e.preventDefault();
			else
				e.returnValue= false;
			return false;
		}
	}
</script>
