    <script>
        var webClient = null;
        var onWebClientMessage = function (message, origin) {
                
                document.querySelector('#webClientMessageArea').innerText = message;
                simulateClick();
                
            };
        var messageToWebClient = function (message) {
            webClient.postMessage(message);
        };
        var init = function () {
            webClient = new WebClient1CE('webClientContainer',
            {
                webClientURL: 'http://84.201.169.9:8080/zenin_webdms_dev?MainWindowMode=EmbeddedWorkplace&debug=tcp&DebuggerURL=84.201.169.9:30205',
                width: '100%',
                height: '100%',
                events:
                {
                    onMessage: onWebClientMessage
                }
            });
        };
        function simulateClick() {
			  var event = new MouseEvent('click', {
			    'view': window,
			    'bubbles': true,
			    'cancelable': true
			  });
			  var cb = document.getElementById('webClientMessageArea'); 
			 cb.dispatchEvent(event);
			};
						
		function createButtons(message){
				document.querySelector('#webClientMessageArea').innerText = message;
                simulateClick();
			}
			      
		function postOnWeb(message){
                messageToWebClient(message)
			}      
		 
    </script>
