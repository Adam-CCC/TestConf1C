CREATE PROCEDURE [dbo].[clear_db]
AS
BEGIN
	BEGIN TRANSACTION;
	SET NOCOUNT ON;	
	DELETE FROM journ_rec;
	DELETE FROM journ;
	DELETE FROM props;	
	DELETE FROM metadata;
	DELETE FROM users;
	DELETE FROM computers;
	COMMIT;
END

CREATE VIEW [dbo].[journ_view]
AS
SELECT     TOP (100) PERCENT dbo.journ.uid, dbo.journ.datechange, dbo.journ.ver, dbo.journ.caption, dbo.computers.caption AS computer, 
                      dbo.users.caption AS userchange, dbo.metadata.caption AS meta, dbo.events.caption AS event, dbo.ribs.caption AS ribcaption, dbo.journ.base_uid
FROM         dbo.journ LEFT OUTER JOIN
                      dbo.computers ON dbo.journ.computer_uid = dbo.computers.uid LEFT OUTER JOIN
                      dbo.users ON dbo.journ.user_uid = dbo.users.uid LEFT OUTER JOIN
                      dbo.metadata ON dbo.journ.meta_uid = dbo.metadata.uid LEFT OUTER JOIN
                      dbo.ribs ON dbo.journ.rib_uid = dbo.ribs.uid LEFT OUTER JOIN
                      dbo.events ON dbo.journ.event_uid = dbo.events.uid
ORDER BY dbo.journ.uid

GO
