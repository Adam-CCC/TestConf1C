TAR = Справочники.Renault_МодулиДляВыполнения.TAR.Ссылка;
СтруктураШапки   = TAR.ПараметрыТаблицы.НайтиСтроки(Новый Структура("Строка", "DDDD"));
СтруктураСтроки  = TAR.ПараметрыТаблицы.НайтиСтроки(Новый Структура("Строка", "TAR"));
СтруктураПодвала = TAR.ПараметрыТаблицы.НайтиСтроки(Новый Структура("Строка", "FFFF"));
сПараметры = Новый Структура;

История.УстановитьПараметр("ТекущаяДата",  ТекущаяДата());

История.УстановитьПараметр("НастройкаGRP", Справочники.Renault_МодулиДляВыполнения.GRP.Ссылка);
История.УстановитьПараметр("НастройкаRDG", Справочники.Renault_МодулиДляВыполнения.RDG.Ссылка);
Выборка = История.Выполнить();

Если Не Выборка.Пустой() Тогда // Есть настройки обмена. Делаем обмен.

    РезультатИстория = Выборка.Выбрать();

    Пока РезультатИстория.Следующий() Цикл

        Если (РезультатИстория.НомерСообщения - РезультатИстория.НомерСообщенияGRP) > 2
                Или (РезультатИстория.НомерСообщения - РезультатИстория.НомерСообщенияRDG) > 2 Тогда

        КонецЕсли;

        Если ТекущаяДата() - РезультатИстория.ДатаСообщения < 3600 Тогда
            Продолжить;
        КонецЕсли;

        ТекстДокумент = Новый ТекстовыйДокумент;
        ИтогоСтрок = 0;

        // Шапка
        СтрокаДокумента = Справочники.Renault_МодулиДляВыполнения.СформироватьСтрокуПоШаблону(РезультатИстория, СтруктураШапки);

        // Строки
        Запрос = Новый Запрос(СтруктураЗапросов.Основной);
        Запрос.УстановитьПараметр("ТипЦены",        Справочники.Renault_ОбщиеНастройкиБазыДанных.ЭтаБаза.ОптоваяЦена);
        Запрос.УстановитьПараметр("ДилерскийЦентр", РезультатИстория.BIR);

        ТЗРезультат = Запрос.Выполнить().Выгрузить();
        Для Каждого СтрокаТЗ из ТЗРезультат Цикл 
            СтрокаТЗ.Артикул = Прав("0000000000" + СтрокаТЗ.Артикул,10);
            СтрокаТЗ.Артикул = Врег(СтрокаТЗ.Артикул);
        КонецЦикла;

            СтрокаДокумента = Справочники.Renault_МодулиДляВыполнения.СформироватьСтрокуПоШаблону(СтрокаТЗ, СтруктураСтроки);
            ТекстДокумент.ДобавитьСтроку(СтрокаДокумента);
            ИтогоСтрок = ИтогоСтрок + 1;
            ПредыдущийАртикул = СтрокаТЗ.Артикул;
        КонецЦикла;

        // Подвал
        СтрокаДокумента = Справочники.Renault_МодулиДляВыполнения.СформироватьСтрокуПоШаблону(РезультатИстория, СтруктураПодвала,ИтогоСтрок);

        Справочники.Renault_МодулиДляВыполнения.ОтправитьТекстовыйДокумертРекурсивно(ТекстДокумент, сПараметры);
