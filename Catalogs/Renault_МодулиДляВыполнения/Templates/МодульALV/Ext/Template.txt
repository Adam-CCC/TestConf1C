ALV = Справочники.Renault_МодулиДляВыполнения.ALV.Ссылка;
СтруктураШапки     = ALV.ПараметрыТаблицы.НайтиСтроки(Новый Структура("Строка", "DDDD"));
СтруктураЗаголовок = ALV.ПараметрыТаблицы.НайтиСтроки(Новый Структура("Строка", "HAL"));
СтруктураСтроки    = ALV.ПараметрыТаблицы.НайтиСтроки(Новый Структура("Строка", "LAL"));
СтруктураПодвала   = ALV.ПараметрыТаблицы.НайтиСтроки(Новый Структура("Строка", "FFFF"));
сПараметры = Новый Структура;

История = Новый Запрос(СтруктураЗапросов.Настройки);
История.УстановитьПараметр("ТекущаяДата", ТекущаяДата());
История.УстановитьПараметр("Настройка",   ALV);
Выборка = История.Выполнить();

Если Не Выборка.Пустой() Тогда // Есть настройки обмена. Делаем обмен.

    РезультатИстория = Выборка.Выбрать();

    Пока РезультатИстория.Следующий() Цикл

        ТекстДокумент = Новый ТекстовыйДокумент;
        ИтогоСтрок = 0;

        // Шапка
        СтрокаДокумента = Справочники.Renault_МодулиДляВыполнения.СформироватьСтрокуПоШаблону(РезультатИстория, СтруктураШапки);
        ТекстДокумент.ДобавитьСтроку(СтрокаДокумента);

        // Документы
        Запрос = Новый Запрос(СтруктураЗапросов.Основной);
        Запрос.УстановитьПараметр("ДилерскийЦентр", РезультатИстория.ДилерскийЦентр); 

        НаборЗаписей = РегистрыСведений.Renault_ЗаказыПокупателейИсторияВыгрузки.СоздатьНаборЗаписей();

        Заголовок = Запрос.Выполнить().Выбрать(ОбходРезультатаЗапроса.ПоГруппировкам);
        Пока Заголовок.Следующий() Цикл

            Если Заголовок.КолСтрок = 0 Тогда // Нечего выгружать
                Продолжить;
            КонецЕсли;

            сЗаголовок = Новый Структура;
            сЗаголовок.Вставить("НомерВРеноПартс",     Заголовок.НомерВРеноПартс);
            сЗаголовок.Вставить("НомерОтправки",       Прав(Заголовок.НомерДокумента, 6) + Формат(РезультатИстория.НомерСообщения,"ЧЦ=4; ЧВН=; ЧГ=0"));
            сЗаголовок.Вставить("НомерДокумента",      Заголовок.НомерДокумента);
            сЗаголовок.Вставить("BIR",                 Заголовок.BIR);
            сЗаголовок.Вставить("Клиент",              Заголовок.Клиент);
            сЗаголовок.Вставить("НаименованиеКлиента", Заголовок.Комментарий);
            сЗаголовок.Вставить("ДатаДокумента",       Заголовок.ДатаДокумента);
            сЗаголовок.Вставить("КолСтрок",            Заголовок.КолСтрок);
            СтрокаДокумента = Справочники.Renault_МодулиДляВыполнения.СформироватьСтрокуПоШаблону(сЗаголовок, СтруктураЗаголовок);
            ТекстДокумент.ДобавитьСтроку(СтрокаДокумента);

            // Строки документа
            Выборка = Заголовок.Выбрать();
            НомерСтроки = 1;

            Пока Выборка.Следующий() Цикл

                Если Выборка.Резерв = Выборка.КолВыгружено и не Выборка.Отмена Тогда // Выгружены все
                    // Обновим историю выгрузки
                    СтрокаНабора = РегистрыСведений.Renault_ЗаказыПокупателейИсторияВыгрузки.СоздатьМенеджерЗаписи();
                    СтрокаНабора.ЗаказПокупателя = Выборка.Заказ;
                    СтрокаНабора.Номенклатура    = ?(Выборка.НоменклатураЗамена <> NULL, Выборка.НоменклатураЗамена, Выборка.Номенклатура);
                    СтрокаНабора.Количество      = Выборка.КолВыгружено;
                    СтрокаНабора.Записать();

                    Продолжить;

                КонецЕсли;

                сСтрока = Новый Структура;
                сСтрока.Вставить("НомерСтр",          Выборка.НомерСтроки);
                сСтрока.Вставить("Артикул",           Прав("0000000000" + СокрЛП(ВРег(Выборка.Артикул)), 10));
                Если ЗначениеЗаполнено(Выборка.АртикулЗамены) Тогда
                    сСтрока.Вставить("АртикулЗамены", Прав("0000000000" + СокрЛП(ВРег(Выборка.АртикулЗамены)), 10));
                Иначе
                    сСтрока.Вставить("АртикулЗамены", "");
                КонецЕсли;
                сСтрока.Вставить("КолВсего",          ?(Выборка.Отмена, Выборка.КолВсего, Выборка.КолВсего - Выборка.КолВыгружено));
                сСтрока.Вставить("КолДоставлено",     ?(Выборка.Отмена, Выборка.Резерв, Выборка.Резерв - Выборка.КолВыгружено));
                Если Выборка.Отмена Тогда
                    СтатусПоставки = "2";
                ИначеЕсли Выборка.КолВсего <= Выборка.Резерв  Тогда
                    СтатусПоставки = "0";
                ИначеЕсли Выборка.КолВсего > Выборка.Резерв Тогда
                    СтатусПоставки = "1";
                КонецЕсли;
                сСтрока.Вставить("СтатусПоставки", СтатусПоставки);

                СтрокаДокумента = Справочники.Renault_МодулиДляВыполнения.СформироватьСтрокуПоШаблону(сСтрока, СтруктураСтроки);
                ТекстДокумент.ДобавитьСтроку(СтрокаДокумента);
                НомерСтроки = НомерСтроки + 1;
                ИтогоСтрок  = ИтогоСтрок + 1;

                // Обновим историю выгрузки
                СтрокаНабора = РегистрыСведений.Renault_ЗаказыПокупателейИсторияВыгрузки.СоздатьМенеджерЗаписи();
                СтрокаНабора.ЗаказПокупателя = Выборка.Заказ;
                СтрокаНабора.Номенклатура    = ?(Выборка.НоменклатураЗамена <> NULL, Выборка.НоменклатураЗамена, Выборка.Номенклатура);
                СтрокаНАбора.Количество      = ?(Выборка.Отмена, Выборка.КолВсего + Выборка.КолВыгружено, Выборка.Резерв);
                СтрокаНабора.Записать();

                Если Выборка.Отмена Тогда
                    // Обновим историю отказов
                    Менеджер = РегистрыСведений.Renault_ОтказыПоЗаказам.СоздатьМенеджерЗаписи();
                    Менеджер.Заказ        = Выборка.Заказ;
                    Менеджер.Номенклатура = Выборка.Номенклатура;
                    Менеджер.Количество   = Выборка.КолВсего;
                    Менеджер.Обработан    = Истина;
                    Менеджер.Записать();
                КонецЕсли;

            КонецЦикла;

            ИтогоСтрок = ИтогоСтрок + 1;

        КонецЦикла;

        Если ИтогоСтрок = 0 Тогда // Строк нет, ALV отправлять не надо
            Продолжить;
        КонецЕсли;

        // Подвал
        СтрокаДокумента = Справочники.Renault_МодулиДляВыполнения.СформироватьСтрокуПоШаблону(РезультатИстория, СтруктураПодвала,ИтогоСтрок);
        ТекстДокумент.ДобавитьСтроку(СтрокаДокумента);

        // Обновляем историю
        Менеджер = РегистрыСведений.Renault_ИсториОбменаССайтом.СоздатьМенеджерЗаписи();
        Менеджер.ДилерскийЦентр = РезультатИстория.ДилерскийЦентр;
        Менеджер.Настройка      = ALV;
        Менеджер.НомерСообщения = РезультатИстория.НомерСообщения;
        Менеджер.ДатаСообщения  = РезультатИстория.ДатаСообщения;
        Менеджер.Записать();

        // Отправляем
        сПараметры.Вставить("БИР", Лев(РезультатИстория.BIR, 8));
        сПараметры.Вставить("Поток", "ALV");
        сПараметры.Вставить("НомерСообщения", Формат(РезультатИстория.НомерСообщения, "ЧЦ=4; ЧВН=; ЧГ=0"));
        Справочники.Renault_МодулиДляВыполнения.ОтправитьТекстовыйДокумент(ТекстДокумент, сПараметры);

    КонецЦикла;
